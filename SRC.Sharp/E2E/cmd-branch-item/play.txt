expect eval a1 = 2

expect eval stock1 = 2
# (2) ユニットＩＤ・ユニット名称・パイロット名で指定して外す

# 味方以外のユニットから外したアイテムは削除され、在庫に残らない

# (3) アイテム番号: デフォルトユニットの２番目を削除する（在庫に戻らない）

expect eval a5 = 0

# 名称: 装備されていないものを優先して削除する
expect eval stock8 = 1

expect eval b3 = 2
